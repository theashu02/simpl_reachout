  private apiKey: string;
  private baseUrl = 'https://google.serper.dev/search';

  constructor(apiKey: string) {
    this.apiKey = apiKey;
  }

  async search(query: string): Promise<SerperSearchResult> {
    const response = await fetch(this.baseUrl, {
      method: 'POST',
      headers: {
        'X-API-KEY': this.apiKey,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({ q: query }),
    });

    if (!response.ok) {
      throw new Error(`Serper API error: ${response.statusText}`);
    }

    return response.json();
  }

  async findCompanyInfo(companyName: string): Promise<CompanyInfo> {
    // Search for LinkedIn profile
    const linkedinQuery = `${companyName} site:linkedin.com/company`;
    const linkedinResults = await this.search(linkedinQuery);
    
    // Search for company logo
    const logoQuery = `${companyName} logo`;
    const logoResults = await this.search(logoQuery);

    // Extract LinkedIn URL
    const linkedinUrl = this.extractLinkedInUrl(linkedinResults);
    
    // Extract logo URL
    const logoUrl = this.extractLogoUrl(logoResults);
    
    // Get website from knowledge graph if available
    const website = logoResults.knowledgeGraph?.website;

    return {
      companyName,
      linkedinUrl,
      logoUrl,
      website,
    };
  }

  private extractLinkedInUrl(results: SerperSearchResult): string | null {
    if (!results.organic || results.organic.length === 0) {
      return null;
    }

    // Find the first LinkedIn company URL
    const linkedinResult = results.organic.find(item => 
      item.link.includes('linkedin.com/company/')
    );

    return linkedinResult?.link || null;
  }

  private extractLogoUrl(results: SerperSearchResult): string | null {
    // Try to get logo from knowledge graph first
    if (results.knowledgeGraph?.imageUrl) {
      return results.knowledgeGraph.imageUrl;
    }

    return null;
  }
}

// Initialize Elysia app
const app = new Elysia()
  .get('/', () => ({
    message: 'Company Info Finder API',
    endpoints: {
      '/api/company/:name': 'Get company LinkedIn and logo',
      '/health': 'Health check'
    }
  }))
  
  .get('/health', () => ({ status: 'ok' }))
  
  .get('/api/company/:name', async ({ params, set }) => {
    try {
      const { name } = params;
      
      if (!name || name.trim() === '') {
        set.status = 400;
        return { error: 'Company name is required' };
      }

      // Get API key from environment
      const apiKey = process.env.SERPER_API_KEY;
      
      if (!apiKey) {
        set.status = 500;
        return { error: 'SERPER_API_KEY not configured' };
      }

      const serper = new SerperService(apiKey);
      const companyInfo = await serper.findCompanyInfo(name);

      return {
        success: true,
        data: companyInfo
      };
    } catch (error) {
      set.status = 500;
      return {
        success: false,
        error: error instanceof Error ? error.message : 'Unknown error'
      };
    }
  })
  
  .post('/api/companies/batch', async ({ body, set }) => {
    try {
      const { companies } = body as { companies: string[] };
      
      if (!Array.isArray(companies) || companies.length === 0) {
        set.status = 400;
        return { error: 'Companies array is required' };
      }

      const apiKey = process.env.SERPER_API_KEY;
      
      if (!apiKey) {
        set.status = 500;
        return { error: 'SERPER_API_KEY not configured' };
      }

      const serper = new SerperService(apiKey);
      
      // Process companies in parallel with rate limiting
      const results = await Promise.all(
        companies.map(async (company) => {
          try {
            return await serper.findCompanyInfo(company);
          } catch (error) {
            return {
              companyName: company,
              linkedinUrl: null,
              logoUrl: null,
              error: error instanceof Error ? error.message : 'Unknown error'
            };
          }
        })
      );

      return {
        success: true,
        data: results
      };
    } catch (error) {
      set.status = 500;
      return {
        success: false,
        error: error instanceof Error ? error.message : 'Unknown error'
      };
    }
  })
  
  .listen(3000);

console.log(
